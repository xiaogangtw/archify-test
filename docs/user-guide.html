想看真的腳本長怎樣？展開合成頁面的 mock 範例

這份獨立示範用假頁面與假會員，不是上方案例的真實登入實作，也不會送產品請求。

import { test, expect } from '../../support/web-test.js';

test('模擬登入後顯示會員首頁', async ({ page }) => {
  await test.step('開啟模擬登入頁', async () => {
    await page.setContent(`
      <main>
        <h1>模擬登入</h1>
        <button onclick="document.querySelector('main').innerHTML=
          '&lt;h1&gt;會員首頁&lt;/h1&gt;&lt;p&gt;測試會員 小明&lt;/p&gt;'">
          模擬登入
        </button>
      </main>`);
    await expect(page.getByRole('heading', { name: '模擬登入' })).toBeVisible();
  });
  await test.step('點擊模擬登入', async () => {
    await page.getByRole('button', { name: '模擬登入' }).click();
  });
  await test.step('核對會員首頁', async () => {
    await expect(page.getByRole('heading', { name: '會員首頁' })).toBeVisible();
    await expect(page.getByText('測試會員 小明')).toBeVisible();
  });
});
